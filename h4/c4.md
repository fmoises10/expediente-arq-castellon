## Nivel 2 - Contenedores

```mermaid
C4Container
    title Tienda con Inventario - Nivel 2: Contenedores

    Person(vendedor, "Vendedor", "Consulta productos y registra ventas")
    Person(administrador, "Administrador", "Gestiona productos, inventario, precios y reportes")

    System_Ext(pago, "Servicio de Pagos", "Servicio externo para procesar pagos")
    System_Ext(notificaciones, "Servicio de Notificaciones", "Servicio externo para enviar avisos")

    System_Boundary(tienda, "Sistema de Tienda con Inventario") {

        Container(web, "Aplicación Web", "HTML / CSS / JavaScript", "Interfaz utilizada por vendedores y administradores")

        Container(api, "Backend / API", "Spring Boot", "Recibe solicitudes HTTP y coordina las operaciones del sistema")

        Container(productos, "Gestión de Productos", "Lógica de negocio", "Permite registrar, consultar y modificar productos")

        Container(inventario, "Gestión de Inventario", "Lógica de negocio", "Controla existencias, entradas y salidas")

        Container(ventas, "Gestión de Ventas", "Lógica de negocio", "Registra ventas y sus detalles")

        Container(pagos, "Gestión de Pagos", "Lógica de negocio", "Coordina el procesamiento de pagos")

        Container(reportes, "Gestión de Reportes", "Lógica de negocio", "Genera reportes de ventas e inventario")

        Container(avisos, "Servicio de Avisos", "Observer", "Notifica eventos importantes del sistema")

        ContainerDb(db, "Base de Datos", "SQLite", "Almacena productos, inventario, ventas y usuarios")
    }

    Rel(vendedor, web, "Utiliza")
    Rel(administrador, web, "Utiliza")

    Rel(web, api, "Envía solicitudes HTTP")

    Rel(api, productos, "Gestiona productos")
    Rel(api, inventario, "Gestiona inventario")
    Rel(api, ventas, "Gestiona ventas")
    Rel(api, pagos, "Procesa pagos")
    Rel(api, reportes, "Solicita reportes")

    Rel(productos, db, "Lee y almacena productos")
    Rel(inventario, db, "Lee y actualiza existencias")
    Rel(ventas, db, "Registra ventas")
    Rel(reportes, db, "Consulta información")

    Rel(ventas, pagos, "Solicita pago")
    Rel(pagos, pago, "Procesa pago mediante adaptador")

    Rel(ventas, avisos, "Notifica eventos de venta")
    Rel(inventario, avisos, "Notifica cambios")
    Rel(avisos, notificaciones, "Envía avisos")
