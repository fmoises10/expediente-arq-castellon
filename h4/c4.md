    title Tienda con Inventario - Nivel 2: Contenedores

    Person(vendedor, "Vendedor", "Consulta productos y registra ventas")

    Person(administrador, "Administrador", "Gestiona productos, stock, precios y reportes")

    System_Ext(pago, "Sistema de Pagos", "Procesa los pagos")

    System_Boundary(tienda, "Sistema de Tienda con Inventario") {

        Container(web, "Aplicación Web", "HTML, CSS, JavaScript", "Interfaz utilizada por vendedores y administradores")

        Container(gestion, "Sistema de Gestión", "Lógica de negocio", "Gestiona productos, inventario, ventas, precios y reportes")

        Container(avisos, "Servicio de Avisos", "Observer", "Notifica eventos importantes del sistema")

        ContainerDb(db, "Base de Datos", "SQLite", "Almacena usuarios, productos, inventario y ventas")

    }

    Rel(vendedor, web, "Utiliza")

    Rel(administrador, web, "Utiliza")

    Rel(web, gestion, "Envía solicitudes")

    Rel(gestion, db, "Lee y almacena información")

    Rel(gestion, avisos, "Notifica eventos")

    Rel(gestion, pago, "Solicita procesamiento de pagos")
